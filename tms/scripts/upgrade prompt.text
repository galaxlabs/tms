```python
import frappe
from tms.utils.bot_settings import get_settings, normalize_lang

# ------------------------------------------------------------
# Multi-language output (EN + UR + AR in ONE message)
# ------------------------------------------------------------
DEFAULT_LANGS = ("en", "ur", "ar")

LANG_LABEL = {
    "en": "🇬🇧 English",
    "ur": "🇵🇰 اردو",
    "ar": "🇸🇦 العربية",
}


def _render_template(tpl: str, **kwargs) -> str:
    if not tpl:
        return ""
    try:
        return tpl.format(**kwargs)
    except Exception:
        return tpl


def _resolve_message_from_meta(messages_meta, key: str, lang: str) -> str | None:
    """
    Supports:
    A) { "en": {key:"..."}, "ur": {...}, "ar": {...} }
    B) { key: {"en":"...","ur":"...","ar":"..."} }
    C) { key: "..." }
    """
    if not messages_meta:
        return None

    # A) by-lang root
    if isinstance(messages_meta, dict) and lang in messages_meta and isinstance(messages_meta.get(lang), dict):
        v = messages_meta.get(lang, {}).get(key)
        if isinstance(v, str) and v.strip():
            return v

    # B/C) by-key root
    if isinstance(messages_meta, dict) and key in messages_meta:
        v = messages_meta.get(key)

        if isinstance(v, dict):
            vv = v.get(lang)
            if isinstance(vv, str) and vv.strip():
                return vv

        if isinstance(v, str) and v.strip():
            return v

    return None


def _get_template(key: str, lang: str, fallback: str | None = None) -> str:
    settings, messages_meta = get_settings()

    tpl = _resolve_message_from_meta(messages_meta, key, lang)

    # Optional: direct field on settings (rare)
    if not tpl and settings and hasattr(settings, key):
        val = getattr(settings, key)
        if isinstance(val, str) and val.strip():
            tpl = val

    return tpl.strip() if tpl and tpl.strip() else (fallback or "")


def _is_trilingual_enabled() -> bool:
    """
    Optional checkbox field: send_all_languages
    If missing -> default ON (as you requested)
    """
    try:
        settings, _ = get_settings()
        if settings and hasattr(settings, "send_all_languages"):
            return bool(int(getattr(settings, "send_all_languages") or 0))
    except Exception:
        pass
    return True


def _find_staff_name_for_inbound(doc) -> str:
    """
    Always greet by Staff.full_name if possible.
    Falls back to WhatsApp profile_name, then "Driver".
    """
    try:
        from tms.utils.whatsapp_utils import normalize_phone

        sender_raw = doc.get("from") or doc.get("from_") or ""
        sender = normalize_phone(sender_raw) or sender_raw

        contact_name = frappe.db.get_value("WhatsApp Contact", {"phone": sender}, "name")
        if contact_name:
            staff = frappe.db.get_value("WhatsApp Contact", contact_name, "staff")
            if staff:
                full_name = frappe.db.get_value("Staff", staff, "full_name")
                if full_name:
                    return full_name.strip()
    except Exception:
        pass

    pn = (getattr(doc, "profile_name", None) or "").strip()
    if pn:
        return pn

    return "Driver"


def _send_outgoing_reply(in_doc, message_text: str):
    """
    Create an Outgoing WhatsApp Message row (queued by your WA sender).
    Matches your dedup logic:
      Outgoing, is_reply=1, reply_to_message_id = inbound.message_id
    """
    to_number = in_doc.get("from") or in_doc.get("from_") or ""
    reply_mid = getattr(in_doc, "message_id", None)

    out = frappe.get_doc(
        {
            "doctype": "WhatsApp Message",
            "type": "Outgoing",
            "content_type": "text",
            "message": message_text,
            "to": to_number,
            "is_reply": 1,
            "reply_to_message_id": reply_mid,
        }
    )
    out.insert(ignore_permissions=True)
    return out.name


def _build_trilingual_message(key: str, fallback: str | None = None, **kwargs) -> str:
    """
    Build ONE message containing EN + UR + AR.
    Also injects {name} + {greet} per language automatically.
    """
    name = (kwargs.get("name") or "").strip() or _find_staff_name_for_inbound(kwargs.get("_doc_for_name"))
    kwargs["name"] = name

    parts = []
    for lg in DEFAULT_LANGS:
        # greet in that language
        greet_tpl = _get_template("greet_prefix", lg, fallback="Hi {name} 👋")
        greet = _render_template(greet_tpl, name=name).strip()

        per_lang_kwargs = dict(kwargs)
        per_lang_kwargs["greet"] = greet

        tpl = _get_template(key, lg, fallback=fallback)
        text = _render_template(tpl, **per_lang_kwargs).strip()
        if not text:
            continue

        label = LANG_LABEL.get(lg, lg)
        parts.append(f"{label}\n{text}")

    if not parts:
        raw = _render_template(fallback or "", **kwargs).strip()
        return raw or "…"

    return "\n\n— — —\n\n".join(parts)


def send_reply(doc, key: str, lang: str | None = None, fallback: str | None = None, **kwargs):
    """
    Main helper used everywhere.

    - Trilingual ON -> one message EN+UR+AR
    - Trilingual OFF -> single language (lang)
    """
    try:
        kwargs["_doc_for_name"] = doc  # internal for name lookup

        if _is_trilingual_enabled():
            msg = _build_trilingual_message(key, fallback=fallback, **kwargs)
            _send_outgoing_reply(doc, msg)
            return

        lg = normalize_lang(lang or "en")
        name = (kwargs.get("name") or "").strip() or _find_staff_name_for_inbound(doc)
        greet_tpl = _get_template("greet_prefix", lg, fallback="Hi {name} 👋")
        kwargs["name"] = name
        kwargs["greet"] = _render_template(greet_tpl, name=name).strip()

        tpl = _get_template(key, lg, fallback=fallback)
        msg = _render_template(tpl, **kwargs).strip() or (fallback or "")
        if msg:
            _send_outgoing_reply(doc, msg)

    except Exception:
        frappe.log_error("WA SEND_REPLY FAIL", frappe.get_traceback())
        return
```

## 2) `apps/tms/tms/utils/whatsapp_bot/flows/trip_flow.py`

```python
import frappe
from frappe.utils import nowdate, now_datetime


def create_trip(driver_name: str, contact, route_name: str, is_return_trip: int = 0) -> "frappe.model.document.Document":
    """
    Create a NEW Trip (no OCR here).
    Used ONLY after route selection (or return trip confirm).
    """
    trip = frappe.new_doc("Trip")
    trip.driver = driver_name
    trip.trip_status = "Scheduled"
    trip.date = nowdate()
    trip.departure = now_datetime()
    trip.trip_route = route_name
    if hasattr(trip, "is_return_trip"):
        trip.is_return_trip = int(is_return_trip or 0)

    # Optional from/to if fields exist and Route has from/to
    try:
        r = frappe.get_doc("Route", route_name)
        if trip.meta.has_field("from_location") and getattr(r, "from_city", None):
            trip.from_location = r.from_city
        if trip.meta.has_field("to_location") and getattr(r, "to_city", None):
            trip.to_location = r.to_city
    except Exception:
        pass

    trip.insert(ignore_permissions=True)

    # Attach to contact (single UPDATE)
    updates = {}
    if hasattr(contact, "current_trip"):
        updates["current_trip"] = trip.name
        contact.current_trip = trip.name

    if updates:
        frappe.db.set_value(contact.doctype, contact.name, updates, update_modified=False)

    return trip


def clone_return_trip(base_trip_name: str, driver_name: str, contact, return_route_name: str) -> str:
    """
    Clone passengers from base trip to a new trip using return_route.
    No OCR/Gemini.
    Returns new trip name.
    """
    base = frappe.get_doc("Trip", base_trip_name)

    new_trip = create_trip(driver_name, contact, return_route_name, is_return_trip=1)

    # Copy passengers
    try:
        new_trip.set("passengers", [])
        for p in (base.get("passengers") or []):
            new_trip.append(
                "passengers",
                {
                    "passenger_name": p.get("passenger_name"),
                    "idpassport_no": p.get("idpassport_no"),
                    "nationality": p.get("nationality"),
                    "contact_no": p.get("contact_no"),
                },
            )
        new_trip.save(ignore_permissions=True)
    except Exception:
        frappe.log_error("RETURN TRIP CLONE PASSENGERS FAIL", frappe.get_traceback())

    return new_trip.name
```

```python
import frappe
from tms.utils.whatsapp_bot.helpers.json_store import get_json, set_json
from tms.utils.whatsapp_bot.helpers.messaging import send_reply
from tms.utils.whatsapp_bot.flows.passenger_flow import write_passengers_and_finalize
from tms.utils.whatsapp_utils import send_trip_pdf_via_whatsapp
from tms.utils.whatsapp_bot.flows.trip_flow import create_trip


def get_available_routes(limit: int = 15) -> list[dict]:
    return (
        frappe.db.get_all(
            "Route",
            fields=["name", "from_city", "to_city"],
            order_by="modified desc",
            limit=limit,
        )
        or []
    )


def _routes_lines(routes: list[dict]) -> str:
    lines = []
    for i, r in enumerate(routes, start=1):
        fc = (r.get("from_city") or "").strip()
        tc = (r.get("to_city") or "").strip()
        lines.append(f"{i}) {fc} → {tc}")
    return "\n".join(lines)


def send_route_list(doc, contact, lang):
    routes = get_available_routes()
    if not routes:
        send_reply(doc, "no_routes_defined", lang, fallback="⚠️ No routes configured.")
        return False

    route_names = [r["name"] for r in routes if r.get("name")]
    set_json(contact, "route", route_names)

    # Store state
    contact.db_set("bot_state", "WAITING_ROUTE", update_modified=False)

    send_reply(
        doc,
        "ask_route",
        lang,
        fallback="🛣️ Please reply with route number:\n{routes}",
        routes=_routes_lines(routes),
    )
    return True


def handle_route_choice(ctx, n) -> bool:
    doc = ctx["doc"]
    contact = ctx["contact"]
    lang = ctx["lang"]
    driver = ctx.get("driver")

    try:
        n_int = int(str(n).strip())
    except Exception:
        return False

    route_names = get_json(contact, "route", []) or []
    if not route_names:
        send_reply(doc, "route_invalid", lang, fallback="❗ Route list missing. Type 'route' to show again.")
        return False

    idx = n_int - 1
    if idx < 0 or idx >= len(route_names):
        return False

    route_name = route_names[idx]
    if not frappe.db.exists("Route", route_name):
        return False

    # ✅ Create Trip NOW (after route confirmed)
    trip = create_trip(driver, contact, route_name, is_return_trip=0)
    ctx["trip"] = trip

    send_reply(doc, "route_selected", lang, fallback="✅ Route selected.")
    return finalize_trip(ctx)


def finalize_trip(ctx) -> bool:
    doc = ctx["doc"]
    contact = ctx["contact"]
    lang = ctx["lang"]
    trip = ctx.get("trip")

    if not trip:
        send_reply(doc, "trip_missing", lang, fallback="❗ Trip not found. Type 'reset' to start again.")
        return False

    added = write_passengers_and_finalize(trip, contact, lang) or 0

    try:
        send_trip_pdf_via_whatsapp(trip.name)
    except Exception:
        frappe.log_error("WA PDF SEND FAIL", frappe.get_traceback())
        send_reply(doc, "pdf_send_failed", lang, fallback="⚠️ Trip created but PDF sending failed.")
        return False

    # mark done
    contact.db_set("bot_state", "DONE", update_modified=False)

    # clear batch link if any
    if hasattr(contact, "active_media_batch"):
        contact.db_set("active_media_batch", None, update_modified=False)

    send_reply(doc, "trip_done", lang, fallback="✅ Trip ready. PDF sent. Passengers added: {n}.", n=added)
    return True
```

## 4) `apps/tms/tms/utils/whatsapp_bot/handlers/text.py`

```python
import re
import json
import frappe
from frappe.utils import nowdate

from tms.utils.whatsapp_bot.helpers.messaging import send_reply
from tms.utils.whatsapp_bot.helpers.json_store import get_json, set_json
from tms.utils.whatsapp_bot.flows.route_flow import handle_route_choice, send_route_list
from tms.utils.whatsapp_bot.flows.trip_flow import clone_return_trip
from tms.utils.bot_settings import get_settings


YES_SET = {"yes", "y", "ok", "okay", "1", "ہاں", "haan", "han", "جي", "نعم"}
NO_SET = {"no", "n", "0", "نہیں", "nahin", "nai", "لا"}


def handle_text(ctx):
    doc = ctx["doc"]
    contact = ctx["contact"]
    lang = ctx["lang"]
    driver = ctx.get("driver")

    text = (doc.message or "").strip()
    t = text.lower().strip()

    if not text:
        send_reply(doc, "ask_passenger_count_invalid", lang, fallback="ℹ️ Please send a message.")
        return

    state = _get_state(contact)

    # -------------------------
    # COMMANDS (always)
    # -------------------------
    if t in ("help", "menu"):
        send_reply(doc, "help_menu", lang, fallback="Type start to begin.")
        return

    if t in ("status",):
        _send_status(doc, contact, lang)
        return

    if t in ("start", "reset"):
        _hard_reset(contact)
        send_reply(doc, "ask_passenger_count", lang, fallback="✅ Started. Send number of passengers (example: 3).")
        return

    if t in ("cancel",):
        _cancel_active_batch(contact)
        contact.db_set("bot_state", "WAITING_START", update_modified=False)
        send_reply(doc, "cancel_ok", lang, fallback="✅ Cancelled. Type start to begin again.")
        return

    # If bot not started yet, require start
    if state in ("", "WAITING_START"):
        # show welcome/help and ask to type start
        send_reply(doc, "welcome_menu", lang, fallback="Type *start* to begin.")
        return

    # -------------------------
    # EXTRA CONFIRM (docs > expected)
    # -------------------------
    if state == "EXTRA_CONFIRM":
        pending = get_json(contact, "received_files_json", {}) or {}
        new_expected = int(pending.get("pending_expected") or 0)
        if not new_expected:
            # fallback: go back to collecting
            contact.db_set("bot_state", "COLLECTING_DOCS", update_modified=False)
            return

        if t in YES_SET:
            _update_expected(contact, new_expected)
            # now run OCR if we already have enough
            send_reply(
                doc,
                "all_docs_received_processing",
                lang,
                fallback="✅ All documents received ({received}/{expected}). Now processing…",
                received=new_expected,
                expected=new_expected,
            )
            # trigger OCR by setting state and letting next media handler run?
            # We call a tiny internal flag and OCR will run on next inbound.
            # Better: send instruction to send any message if you want immediate OCR,
            # but we can also set flag 'force_ocr' and media handler will do OCR on next inbound.
            pending["force_ocr"] = 1
            set_json(contact, "received_files_json", pending)
            contact.db_set("bot_state", "COLLECTING_DOCS", update_modified=False)
            return

        if t in NO_SET:
            # ignore extras: restore expected to original, OCR will run on first expected
            pending.pop("pending_expected", None)
            pending.pop("force_ocr", None)
            set_json(contact, "received_files_json", pending)
            contact.db_set("bot_state", "COLLECTING_DOCS", update_modified=False)

            send_reply(
                doc,
                "all_docs_received_processing",
                lang,
                fallback="✅ All documents received ({received}/{expected}). Now processing…",
                received=int(getattr(contact, "expected_passengers", 0) or 0),
                expected=int(getattr(contact, "expected_passengers", 0) or 0),
            )

            # OCR will run when media handler sees received>=expected and not yet OCRed
            return

        # invalid answer
        send_reply(
            doc,
            "unknown_state",
            lang,
            fallback="Reply YES or NO.",
        )
        return

    # -------------------------
    # RETURN FLOW (today)
    # -------------------------
    if t in ("return", "ret", "backtrip"):
        _start_return_confirm(doc, contact, lang, driver)
        return

    if state == "RETURN_CONFIRM":
        pending = get_json(contact, "received_files_json", {}) or {}
        base_trip = pending.get("pending_return_base_trip")
        return_route = pending.get("pending_return_route")

        if t in YES_SET and base_trip and return_route:
            try:
                new_trip_name = clone_return_trip(base_trip, driver, contact, return_route)
                # mark done
                contact.db_set("bot_state", "DONE", update_modified=False)
                pending.pop("pending_return_base_trip", None)
                pending.pop("pending_return_route", None)
                set_json(contact, "received_files_json", pending)
                send_reply(
                    doc,
                    "trip_done",
                    lang,
                    fallback="✅ Return trip created. PDF will be sent.",
                    n=0,
                )
                # send pdf
                try:
                    from tms.utils.whatsapp_utils import send_trip_pdf_via_whatsapp
                    send_trip_pdf_via_whatsapp(new_trip_name)
                except Exception:
                    frappe.log_error("RETURN PDF SEND FAIL", frappe.get_traceback())
                return
            except Exception:
                frappe.log_error("RETURN TRIP CREATE FAIL", frappe.get_traceback())
                send_reply(doc, "unknown_state", lang, fallback="⚠️ Failed to create return trip. Type reset.")
                return

        if t in NO_SET:
            pending.pop("pending_return_base_trip", None)
            pending.pop("pending_return_route", None)
            set_json(contact, "received_files_json", pending)
            contact.db_set("bot_state", "DONE", update_modified=False)
            send_reply(doc, "cancel_ok", lang, fallback="✅ Cancelled.")
            return

        send_reply(doc, "unknown_state", lang, fallback="Reply YES or NO.")
        return

    # -------------------------
    # ROUTE selection
    # -------------------------
    if state == "WAITING_ROUTE":
        n = _extract_digits_1_2(text)
        if n is None:
            send_reply(doc, "route_invalid", lang, fallback="❗ Please reply with the route number (example: 2).")
            return

        ok = handle_route_choice(ctx, n)
        if not ok:
            send_reply(doc, "route_invalid", lang, fallback="❗ Please reply with the route number (example: 2).")
        return

    # -------------------------
    # Passenger count (ONLY after start)
    # -------------------------
    if state == "WAITING_PASSENGER_COUNT":
        n = _extract_digits_1_2(text)
        if n is None or n <= 0:
            send_reply(doc, "ask_passenger_count_invalid", lang, fallback="ℹ️ Send number of passengers only (example: 3).")
            return

        settings, _meta = get_settings()
        max_p = int(getattr(settings, "max_passengers", 0) or 0)
        if max_p and n > max_p:
            send_reply(doc, "max_passengers_exceeded", lang, fallback="❗ Max passengers allowed is {max}.", max=max_p)
            return

        _start_batch_for_count(contact, int(n))

        # ✅ one confirm (count saved) and instruction (send docs)
        send_reply(
            doc,
            "passenger_count_saved",
            lang,
            fallback="✅ Passenger count saved: {expected}. Now send {expected} documents (one per passenger).",
            expected=int(n),
        )
        return

    # -------------------------
    # Collecting docs: ignore text (use status/help)
    # -------------------------
    if state == "COLLECTING_DOCS":
        send_reply(doc, "status_hint", lang, fallback="Send remaining documents or type status.")
        return

    if state == "OCR_PROCESSING":
        send_reply(doc, "ocr_processing", lang, fallback="⏳ Processing documents… please wait.")
        return

    if state == "RESENDING_DOCS":
        send_reply(doc, "resend_done_wait", lang, fallback="✅ Please resend requested passenger documents.")
        return

    if state == "DONE":
        send_reply(doc, "welcome_menu", lang, fallback="✅ Done. Type start for new trip.")
        return

    send_reply(doc, "unknown_state", lang, fallback="ℹ️ Type reset to start again.")


# -----------------------------
# Helpers
# -----------------------------
def _get_state(contact) -> str:
    return (getattr(contact, "bot_state", "") or "").strip().upper()


def _hard_reset(contact):
    """
    Hard reset: cancel batch, clear trip pointers, go back to WAITING_PASSENGER_COUNT.
    """
    _cancel_active_batch(contact)

    frappe.db.set_value(
        contact.doctype,
        contact.name,
        {
            "bot_state": "WAITING_PASSENGER_COUNT",
            "expected_passengers": 0,
            "received_images": 0,
            "resend_ptr": 0,
            "current_trip": None,
            "collected_file_urls_json": "[]",
            "resend_indexes_json": "[]",
            "received_files_json": "{}",
        },
        update_modified=False,
    )

    contact.bot_state = "WAITING_PASSENGER_COUNT"
    contact.expected_passengers = 0
    contact.received_images = 0
    contact.resend_ptr = 0
    contact.current_trip = None
    contact.collected_file_urls_json = "[]"
    contact.resend_indexes_json = "[]"
    contact.received_files_json = "{}"

    set_json(contact, "route", [])


def _cancel_active_batch(contact):
    """
    If WhatsApp Media Batch is used, mark it cancelled and unlink.
    (Safe even if doctype not used yet.)
    """
    try:
        b = getattr(contact, "active_media_batch", None)
        if b and frappe.db.exists("WhatsApp Media Batch", b):
            frappe.db.set_value("WhatsApp Media Batch", b, {"status": "CANCELLED"}, update_modified=False)
    except Exception:
        pass

    if hasattr(contact, "active_media_batch"):
        try:
            contact.db_set("active_media_batch", None, update_modified=False)
        except Exception:
            pass


def _start_batch_for_count(contact, expected: int):
    """
    Create/replace active WhatsApp Media Batch.
    Also reset contact json storage.
    """
    _cancel_active_batch(contact)

    batch_name = None
    try:
        if frappe.db.exists("DocType", "WhatsApp Media Batch"):
            batch = frappe.new_doc("WhatsApp Media Batch")
            batch.contact = contact.name
            # staff link if available
            if getattr(contact, "staff", None):
                batch.staff = contact.staff
            batch.expected_passengers = expected
            batch.received_count = 0
            batch.status = "COLLECTING"
            batch.insert(ignore_permissions=True)
            batch_name = batch.name
    except Exception:
        frappe.log_error("WA BATCH CREATE FAIL", frappe.get_traceback())

    updates = {
        "expected_passengers": expected,
        "received_images": 0,
        "resend_ptr": 0,
        "bot_state": "COLLECTING_DOCS",
        "collected_file_urls_json": "[]",
        "resend_indexes_json": "[]",
        "received_files_json": "{}",
    }
    if hasattr(contact, "active_media_batch"):
        updates["active_media_batch"] = batch_name

    frappe.db.set_value(contact.doctype, contact.name, updates, update_modified=False)

    # in-memory sync
    contact.expected_passengers = expected
    contact.received_images = 0
    contact.resend_ptr = 0
    contact.bot_state = "COLLECTING_DOCS"

    """

    """

    if _get_state(contact) != "DONE":

        order_by="modified desc",
    )

    if not base_trip:
        send_reply(doc, "unknown_state", lang, fallback="No trip found for today.")

        return

        send_reply(doc, "unknown_state", lang, fallback="Return route not available for this route.")

        doc,
        "unknown_state",
        lang,

    )
```

```python
import frappe
from tms.utils.whatsapp_bot.helpers.json_store import get_json, set_json
from tms.utils.whatsapp_bot.helpers.messaging import send_reply
from tms.utils.passenger_ocr_service import process_passenger_images_batch


def handle_media(ctx):
    """
    Bulk-safe media intake:
    - No per-image spam.
    - When received == expected -> send ONE confirm + OCR once.
    - If extra -> ask to increase count (YES/NO).
    """
    doc = ctx["doc"]
    contact = ctx["contact"]
    lang = ctx["lang"]

    state = (getattr(contact, "bot_state", "") or "").strip().upper()
    file_url = (getattr(doc, "attach", "") or "").strip()
    if not file_url:
        return

    # Bot must be started
    if state in ("", "WAITING_START"):
        send_reply(doc, "welcome_menu", lang, fallback="Type *start* to begin.")
        return

    # Must have passenger count first
    expected = int(getattr(contact, "expected_passengers", 0) or 0)
    if expected <= 0 or state == "WAITING_PASSENGER_COUNT":
        send_reply(doc, "ask_passenger_count", lang, fallback="Send passenger count first (example: 3).")
        return

    if state == "OCR_PROCESSING":
        send_reply(doc, "ocr_processing", lang, fallback="⏳ Processing documents… please wait.")
        return

    # Load urls list
    urls = get_json(contact, "collected_file_urls_json", []) or []

    # Dedup retries (only in normal collecting)
    if state not in ("RESENDING_DOCS",) and file_url in urls:
        return

    # -------------------------
    # RESEND mode (replace specific index)
    # -------------------------
    if state == "RESENDING_DOCS":
        resend = get_json(contact, "resend_indexes_json", []) or []
        ptr = int(getattr(contact, "resend_ptr", 0) or 0)

        if not resend or ptr >= len(resend):
            # Safety: rerun OCR with what we have
            return _run_ocr(doc, contact, lang)

        n = int(resend[ptr])
        idx0 = n - 1

        while len(urls) < max(expected, idx0 + 1):
            urls.append("")

        urls[idx0] = file_url
        set_json(contact, "collected_file_urls_json", urls)

        new_ptr = ptr + 1
        frappe.db.set_value(contact.doctype, contact.name, {"resend_ptr": new_ptr}, update_modified=False)
        contact.resend_ptr = new_ptr

        # ask next resend only (no extra noise)
        if new_ptr < len(resend):
            next_n = int(resend[new_ptr])
            send_reply(doc, "resend_document", lang, fallback="⚠️ Resend passenger #{n}.", n=next_n)
            return

        # done resends -> OCR
        return _run_ocr(doc, contact, lang)

    # -------------------------
    # Normal collecting
    # -------------------------
    urls.append(file_url)
    set_json(contact, "collected_file_urls_json", urls)

    received = len([u for u in urls if u])
    frappe.db.set_value(contact.doctype, contact.name, {"received_images": received}, update_modified=False)
    contact.received_images = received

    # Update batch received_count if exists
    try:
        b = getattr(contact, "active_media_batch", None)
        if b and frappe.db.exists("WhatsApp Media Batch", b):
            frappe.db.set_value("WhatsApp Media Batch", b, {"received_count": received}, update_modified=False)
    except Exception:
        pass

    # EXTRA docs
    if received > expected:
        pending = get_json(contact, "received_files_json", {}) or {}
        pending["pending_expected"] = received
        set_json(contact, "received_files_json", pending)

        contact.db_set("bot_state", "EXTRA_CONFIRM", update_modified=False)

        # ✅ ask only about COUNT (not images one-by-one)
        send_reply(
            doc,
            "unknown_state",
            lang,
            fallback="⚠️ Extra document received. You sent {received} docs but passenger count is {expected}. Increase passenger count to {received}? Reply YES or NO.",
            received=received,
            expected=expected,
        )
        return

    # If not complete, stay silent (bulk friendly)
    if received < expected:
        return

    # received == expected -> one confirm + OCR once
    send_reply(
        doc,
        "all_docs_received_processing",
        lang,
        fallback="✅ All documents received ({received}/{expected}). Now processing…",
        received=received,
        expected=expected,
    )
    return _run_ocr(doc, contact, lang)


def _run_ocr(doc, contact, lang):
    expected = int(getattr(contact, "expected_passengers", 0) or 0)
    urls = (get_json(contact, "collected_file_urls_json", []) or [])[:expected]

    # Mark processing
    frappe.db.set_value(contact.doctype, contact.name, {"bot_state": "OCR_PROCESSING"}, update_modified=False)
    contact.bot_state = "OCR_PROCESSING"

    # ✅ IMPORTANT: no trip_name here (Trip created after route)
    res = process_passenger_images_batch(
        trip_name=None,
        file_urls=urls,
        waba_message=doc.name,
    )

    # Store passengers JSON for later trip insertion
    if res.get("passengers") is not None:
        set_json(contact, "received_files_json", {"passengers": res.get("passengers") or []})

    # OCR OK -> route selection
    if res.get("ok"):
        set_json(contact, "resend_indexes_json", [])
        frappe.db.set_value(
            contact.doctype,
            contact.name,
            {"resend_ptr": 0, "bot_state": "WAITING_ROUTE"},
            update_modified=False,
        )
        contact.resend_ptr = 0
        contact.bot_state = "WAITING_ROUTE"

        from tms.utils.whatsapp_bot.flows.route_flow import send_route_list
        send_route_list(doc, contact, lang)
        return

    # OCR NOT OK -> resend flow
    resend = res.get("resend_indexes") or []
    set_json(contact, "resend_indexes_json", resend)

    frappe.db.set_value(
        contact.doctype,
        contact.name,
        {"resend_ptr": 0, "bot_state": "RESENDING_DOCS"},
        update_modified=False,
    )
    contact.resend_ptr = 0
    contact.bot_state = "RESENDING_DOCS"

    n = int(resend[0]) if resend else 1
    send_reply(
        doc,
        "resend_document",
        lang,
        fallback="⚠️ Passenger #{n} document is not clear. Please resend passenger #{n}.",
        n=n,
    )
```

import frappe
from frappe.utils import now_datetime
from datetime import timedelta


def auto_mark_arrived():
    """
    Mark trips as Arrived if:
      now >= departure + duration_minutes
    Updates:
      trip_status = "Arrived"
      arrived_at = now
    Runs safely/idempotent.
    """
    now = now_datetime()

    # Pull trips that can be auto-completed
    rows = frappe.db.get_all(
        "Trip",
        filters={
            "trip_status": ["in", ["Scheduled", "Departed"]],
        },
        fields=["name", "departure", "duration_minutes", "arrived_at"],
        limit_page_length=500,
        order_by="modified asc",
    ) or []

    for r in rows:
        try:
            if r.get("arrived_at"):
                continue

            dep = r.get("departure")
            dur = r.get("duration_minutes")

            if not dep or not dur:
                continue

            dur = int(dur)
            if dur <= 0:
                continue

            due = dep + timedelta(minutes=dur)

            if now >= due:
                frappe.db.set_value(
                    "Trip",
                    r["name"],
                    {"trip_status": "Arrived", "arrived_at": now},
                    update_modified=False,
                )
        except Exception:
            frappe.log_error("AUTO ARRIVED FAIL", frappe.get_traceback())
            continue
